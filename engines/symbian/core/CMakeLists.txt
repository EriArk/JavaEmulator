cmake_minimum_required(VERSION 3.22)
project(AbyssSymbian LANGUAGES C CXX)
add_compile_definitions(_LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION=1)
if(NOT EXISTS "${EKA2L1_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Prepare the pinned EKA2L1 checkout before enabling symbianExperiment")
endif()
if(EXISTS "${EKA2L1_SOURCE_DIR}/ABYSS_REVISION")
    file(READ "${EKA2L1_SOURCE_DIR}/ABYSS_REVISION" EKA2L1_REVISION)
    string(STRIP "${EKA2L1_REVISION}" EKA2L1_REVISION)
else()
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${EKA2L1_SOURCE_DIR}"
        OUTPUT_VARIABLE EKA2L1_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
if(NOT EKA2L1_REVISION STREQUAL "11f5b6cd141c0884a130ee45743f1f5e17bb1783")
    message(FATAL_ERROR "Unexpected EKA2L1 revision: ${EKA2L1_REVISION}")
endif()
if(EXISTS "${EKA2L1_SOURCE_DIR}/abyss-libuv/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_LIBUV "${EKA2L1_SOURCE_DIR}/abyss-libuv" CACHE PATH "Bundled libuv source")
endif()
add_subdirectory("${EKA2L1_SOURCE_DIR}" eka2l1)
include(patches.cmake)
# Upstream's native patch mode still copies this directory with Lua disabled.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin/scripts")
set_target_properties(native-lib PROPERTIES OUTPUT_NAME abysssymbian)
target_sources(native-lib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/host.cpp")
target_sources(native-lib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/removal.cpp")
target_link_libraries(native-lib PRIVATE archive_static)
