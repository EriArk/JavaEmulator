# Shared checkout acquisition is serialized across ABIs. Source ZIP builds still
# work: only these third-party checkouts require Git, not the AbyssME checkout.
file(MAKE_DIRECTORY "${PEER_SOURCE_ROOT}")
file(LOCK "${PEER_SOURCE_ROOT}/acquire.lock" GUARD PROCESS TIMEOUT 300)
function(peer_command)
    execute_process(COMMAND ${ARGN} RESULT_VARIABLE result)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Peer dependency command failed (${result})")
    endif()
endfunction()
function(peer_source name url revision)
    set(path "${PEER_SOURCE_ROOT}/${name}")
    if(NOT EXISTS "${path}")
        if(PEER_OFFLINE)
            message(FATAL_ERROR "Offline peer source missing: ${name}")
        endif()
        peer_command(${GIT_EXECUTABLE} clone --no-checkout --filter=blob:none "${url}" "${path}")
        peer_command(${GIT_EXECUTABLE} -C "${path}" checkout --detach "${revision}")
    endif()
    execute_process(COMMAND ${GIT_EXECUTABLE} -C "${path}" rev-parse HEAD
        OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual STREQUAL revision)
        message(FATAL_ERROR "Unexpected ${name} revision: ${actual}; expected ${revision}")
    endif()
    peer_command(${GIT_EXECUTABLE} -C "${path}" diff --exit-code HEAD --)
endfunction()
peer_source(libdatachannel https://github.com/paullouisageneau/libdatachannel.git 6b1e2e620f1e37f0eafeee702eaea0043cb305fd)
peer_source(mbedtls https://github.com/Mbed-TLS/mbedtls.git 068ff080b369adfac81509f9b57b2afabaf82dc5)
if(PEER_OFFLINE)
    foreach(sub libdatachannel/deps/libjuice libdatachannel/deps/plog libdatachannel/deps/usrsctp mbedtls/framework)
        if(NOT EXISTS "${PEER_SOURCE_ROOT}/${sub}/.git")
            message(FATAL_ERROR "Offline peer submodule missing: ${sub}")
        endif()
    endforeach()
else()
    peer_command(${GIT_EXECUTABLE} -C "${PEER_SOURCE_ROOT}/libdatachannel" submodule update --init --depth 1 deps/libjuice deps/plog deps/usrsctp)
    peer_command(${GIT_EXECUTABLE} -C "${PEER_SOURCE_ROOT}/mbedtls" submodule update --init --depth 1 framework)
endif()
